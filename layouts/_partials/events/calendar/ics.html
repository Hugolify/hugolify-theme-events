{{- /* Builds the iCalendar file of an event (RFC 5545) and returns it as an `event.ics` resource published next to the event page, e.g. /events/winter-concert/event.ics Hugo only writes the file once something renders its permalink, as the "add to calendar" menu does, see events/calendar.html. Text values are escaped with func/EscapeVCard.html: iCalendar escapes them the way vCard does (RFC 5545, section 3.3.11). The UID is the page URL, so importing the file again updates the event rather than doubling it. @context {map} . The event, as returned by events/calendar/event.html. @return {resource} The iCalendar resource. @example {{ with partial "events/calendar/event.html" . }} {{ (partial "events/calendar/ics.html" .).RelPermalink }} {{ end }} */ -}} {{- $page := .page -}} {{- $escape := "func/EscapeVCard.html" -}} {{- /* A day is a date, a time is a date-time: DTSTART;VALUE=DATE:20261204 */ -}} {{- $type := cond .allday ";VALUE=DATE" "" -}} {{- $stamp := $page.Lastmod -}} {{- if $stamp.IsZero }}{{ $stamp = now }}{{ end -}} {{- $lines := slice "BEGIN:VCALENDAR" "VERSION:2.0" "PRODID:-//Hugolify//hugolify-theme-events//EN" "CALSCALE:GREGORIAN" "METHOD:PUBLISH" "BEGIN:VEVENT" (printf "UID:%s" $page.Permalink) (printf "DTSTAMP:%s" ($stamp.UTC.Format "20060102T150405Z")) (printf "DTSTART%s:%s" $type .start) (printf "DTEND%s:%s" $type .end) (printf "SUMMARY:%s" (partial $escape .title)) (printf "DESCRIPTION:%s" (partial $escape .description)) -}} {{- with .location -}} {{- $lines = $lines | append (printf "LOCATION:%s" (partial $escape .)) -}} {{- end -}} {{- $lines = $lines | append (printf "URL:%s" $page.Permalink) "END:VEVENT" "END:VCALENDAR" -}} {{- /* Publish next to the page, without the baseURL sub-path */ -}} {{- $base := (urls.Parse site.BaseURL).Path | strings.TrimSuffix "/" -}} {{- $target := printf "%sevent.ics" (strings.TrimPrefix $base $page.RelPermalink) -}} {{- return resources.FromString $target (printf "%s\r\n" (delimit $lines "\r\n")) -}}
